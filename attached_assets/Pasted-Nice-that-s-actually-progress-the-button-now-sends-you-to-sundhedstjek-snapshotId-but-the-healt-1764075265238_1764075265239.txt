export function useHealthCheck(options: { snapshotId?: string }) {

  return useQuery({
    queryKey: ["healthCheck", snapshotId ?? "overview"],
    queryFn: async () => {
      if (snapshotId) {

          `/api/policy-snapshots/${snapshotId}/health-check`
