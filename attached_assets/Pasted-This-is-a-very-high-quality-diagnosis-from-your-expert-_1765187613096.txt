#### 1\. Update `vite.config.ts`

This forces Vite to build your frontend into a specific folder inside your server directory, so the server always knows where to look.

```typescript
import { defineConfig } from "vite";
import react from "@vitejs/plugin-react";
import path from "path";

export default defineConfig({
  plugins: [react()],
  base: "/", 
  build: {
    // Output directly to server/public so Express can find it easily
    outDir: "server/public",
    assetsDir: "assets",
    emptyOutDir: true,
    sourcemap: true,
  },
  resolve: {
    alias: {
      "@": path.resolve(__dirname, "./src"),
    },
  },
});
```

#### 2\. Update your Server Entry (e.g., `server/index.ts` or `server.js`)

This ensures static files are served *before* the API or the "catch-all" handler.

```typescript
import express from "express";
import path from "path";
import fs from "fs";

const app = express();

// 1. Define the standardized path
const publicDir = path.resolve(process.cwd(), "server/public");
const indexHtml = path.join(publicDir, "index.html");

// 2. Fail Fast: If build is missing, stop the server immediately so you see the error in logs
if (!fs.existsSync(indexHtml)) {
  console.error("CRITICAL ERROR: Build not found!");
  console.error(`Expected: ${indexHtml}`);
  console.error("Did you run 'npm run build' before starting?");
  process.exit(1); 
}

// 3. Serve Static Assets FIRST (Immutable cache for hashed assets)
app.use("/assets", express.static(path.join(publicDir, "assets"), { 
  immutable: true, 
  maxAge: "1y" 
}));

// 4. Serve other static files (favicon, robots.txt)
app.use(express.static(publicDir, { maxAge: "1h" }));

// ... (Your API routes go here, e.g., app.use('/api', apiRouter)) ...

// 5. SPA Fallback (Only for HTML requests, never for assets)
app.get("*", (req, res, next) => {
  if (req.method !== "GET") return next();
  // Don't serve HTML for missing images/js/css
  if (req.path.startsWith("/assets") || req.path.startsWith("/api")) {
    return res.status(404).json({ error: "Not found" });
  }
  res.sendFile(indexHtml);
});

const PORT = process.env.PORT || 5000;
app.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
});
```
