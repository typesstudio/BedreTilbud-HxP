Pseudocode:

app.get("/api/health-check/user/:userId/overview", async (req, res) => {
  const userId = req.params.userId;

  const snapshots = await PolicySnapshotService.getSnapshotsForUser(userId);
  // If getSnapshotsForUser doesn't exist, implement it using Drizzle

  const policies: HealthCheckOverviewPolicyDto[] = snapshots.map((row) => {
    const hc = row.healthCheckJson as HealthCheckJson | null;

    return {
      id: row.id,
      type: row.type,
      company: row.company ?? null,
      annualPremium: hc?.annualPremium ?? null,
      healthScore: hc?.score ?? null,
      potentialSavingsAnnual: hc?.potentialSavingsAnnual ?? null,
      statusLabel: hc?.statusLabel ?? null,
    };
  });

  const response: HealthCheckOverviewResponse = {
    userId,
    policies,
  };

  return res.json(response);
});

app.get("/api/health-check/policy/:policyId", async (req, res) => {
  const policyId = req.params.policyId;

  const snapshot = await PolicySnapshotService.getSnapshotById(policyId);

  if (!snapshot || !snapshot.healthCheckJson) {
    return res.status(404).json({ error: "Health check not ready" });
  }

  const hc = snapshot.healthCheckJson as HealthCheckJson;
  const response: HealthCheckDetailResponse = hc;

  return res.json(response);
});

app.get("/api/comparisons/:comparisonId/overview", async (req, res) => {
  const comparisonId = req.params.comparisonId;

  const pairs = await ComparisonService.getPairsForComparison(comparisonId);

  if (!pairs.length) {
    return res.status(404).json({ error: "Comparison not found" });
  }

  let totalAnnualCurrent = 0;
  let totalAnnualOffer = 0;

  const policies: ComparisonOverviewPolicyDto[] = pairs.map((pair) => {
    const cj = pair.comparisonJson as ComparisonJson | null;
    const summary = cj?.summary;

    const currentAnnual = summary?.annualCurrent ?? 0;
    const offerAnnual = summary?.annualOffer ?? 0;
    const annualSavings = summary?.annualSavings ?? (currentAnnual - offerAnnual);
    const savingsPercent = summary?.savingsPercent ?? (
      currentAnnual > 0 ? (annualSavings / currentAnnual) * 100 : 0
    );
    const keyHighlights = cj?.highlights ?? [];

    totalAnnualCurrent += currentAnnual;
    totalAnnualOffer += offerAnnual;

    return {
      pairId: pair.id,
      currentPolicyId: pair.currentPolicyId,
      offerPolicyId: pair.offerPolicyId,
      type: pair.type,
      currentAnnual,
      offerAnnual,
      annualSavings,
      savingsPercent,
      keyHighlights,
    };
  });

  const response: ComparisonOverviewResponse = {
    comparisonId,
    totalAnnualCurrent,
    totalAnnualOffer,
    totalAnnualSavings: totalAnnualCurrent - totalAnnualOffer,
    policies,
  };

  return res.json(response);
});


Again: only reading from comparisonJson.
