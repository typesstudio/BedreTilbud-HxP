```ts
  summary_json: SingleHealthCheckSummary;
}
```

### 3.2. New endpoint

Extend your existing route from:

```ts
GET /api/policies/health-check/:snapshotId
```

to one of:

* **Option A (recommended)**
  `GET /api/policies/health-check-overview/:snapshotGroupId`

* **Option B (via a single snapshot)**
  `GET /api/policies/health-check-overview/by-snapshot/:snapshotId` → backend looks up `snapshot_group_id` from that snapshot.

Implementation sketch (Option B):

```ts
// routes.ts

app.get("/api/policies/health-check-overview/by-snapshot/:snapshotId", async (req, res) => {
  const snapshotId = req.params.snapshotId;

  // 1) Find the group
  const snapshot = await db.query.healthChecks.findFirst({
    where: eq(healthChecks.id, snapshotId),
  });

  if (!snapshot) {
    return res.status(404).json({ error: "Snapshot not found" });
  }

  const groupId = snapshot.snapshot_group_id;

  // 2) Get all siblings in same group
  const siblings = await db.query.healthChecks.findMany({
    where: eq(healthChecks.snapshot_group_id, groupId),
  });

  // 3) Parse summaries
  const summaries: SingleHealthCheckSummary[] = siblings
    .map((s) => s.summary_json as SingleHealthCheckSummary)
    .filter(Boolean);

  // 4) Aggregate
  const totalAnnualSavings = summaries.reduce(
    (acc, s) => acc + (s.annualSavings || 0),
    0
  );

  const totalCurrentPremiumYear = summaries.reduce(
    (acc, s) => acc + (s.currentPremiumYear || 0),
    0
  );

  const totalSavingsPct =
    totalCurrentPremiumYear > 0
      ? (totalAnnualSavings / totalCurrentPremiumYear) * 100
      : null;

  const aggregatedIssues = summaries.flatMap((s) =>
    s.issues.map((i) => ({
      ...i,
      policyTypes: [s.policyType],
    }))
  );

  const payload: HealthCheckOverview = {
    totalAnnualSavings,
    totalCurrentPremiumYear,
    totalSavingsPct,
    coverageStatus: summaries,
    aggregatedIssues,
  };

  res.json(payload);
});
```

Edge cases:

* If some siblings do **not** have a health check yet → either exclude them, or include a row with `recommendation: "pending"` and label “Afventer analyse”.
* If `totalCurrentPremiumYear` is missing, show only the DKK amount and hide the “20.5% lavere omkostning” subtitle.
