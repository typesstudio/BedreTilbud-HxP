function useCompanyComparison(id: string) {
  return useQuery({
    queryKey: ["companyComparison", id],
    queryFn: async () => {
      const res = await fetch(`/api/comparisons/${id}`);
      if (!res.ok) throw new Error("Failed to load comparison");
      return res.json();
    },
  });
}
