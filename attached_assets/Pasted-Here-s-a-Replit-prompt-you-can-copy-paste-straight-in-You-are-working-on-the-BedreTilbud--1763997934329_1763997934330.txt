`client/src/utils/transformComparison.ts`:

```ts
import {
  ComparisonViewModel,
  ComparisonPolicyRow,
} from "@/components/comparison/ComparisonPageLayout";
import {
  FeatherHome,
  FeatherBuilding,
  FeatherShield,
  FeatherCar,
  FeatherPlane,
} from "@subframe/core";

const policyIconForType: Record<string, React.ReactNode> = {
  indbo: <FeatherHome />,
  hus: <FeatherBuilding />,
  ulykke: <FeatherShield />,
  bil: <FeatherCar />,
  rejse: <FeatherPlane />,
};

export function transformCompanyComparisonToViewModel(apiData: any): ComparisonViewModel {
  const overall = apiData.overall ?? {};
  const policies = (apiData.policies ?? []) as any[];

  const policyRows: ComparisonPolicyRow[] = policies.map((p) => {
    const icon = policyIconForType[p.policyType] ?? <FeatherHome />;

    return {
      policyType: p.policyType,
      label: p.label ?? p.policyType,
      icon,
      currentAnnual: p.currentAnnual ?? null,
      offerAnnual: p.offerAnnual ?? null,
      annualSavings: p.annualSavings ?? null,
      statusLabel: "Se detaljer",
      statusVariant: "success",
    };
  });

  return {
    title: "Sammenlign forsikrings tilbud",
    subtitle: `Sammenlign dit nuværende tilbud med ${apiData.offerCompany ?? ""}`,
    currentCompanyName: apiData.currentCompany ?? "",
    offerCompanyName: apiData.offerCompany ?? "",
    totalAnnualSavings: overall.annualSavings ?? 0,
    totalSavingsPercent: overall.savingsPercent ?? 0,
    totalCurrentAnnual: overall.totalCurrentAnnualPremium ?? 0,
    totalOfferAnnual: overall.totalOfferAnnualPremium ?? 0,
    policies: policyRows,
    coverageRows: [], // can be filled from apiData.policyComparisons later
    highlights: [],   // same
    cumulativeSavings: undefined,
    chartData: undefined,
    canOpenMessages: !!apiData.threadId,
    messageCtaLabel: "Se beskeder",
  };
}
```

## 5. Apply the same pattern to `policy_snapshots` comparison

We already have `usePolicyComparisons` and `/api/policies/comparisons`.
Create another transformer to map that API into `ComparisonViewModel` (e.g. sum premiums to fill totals and set one row per policy type).

Example:

```ts
export function transformPolicyComparisonsToViewModel(apiData: any): ComparisonViewModel {
  const rows = apiData.comparisons ?? [];

  let totalCurrent = 0;
  let totalOffer = 0;
  rows.forEach((row: any) => {
    const current = row.current?.pricing?.annualPremium ?? 0;
    const bestOffer = row.offers?.[0]?.pricing?.annualPremium ?? 0;
    totalCurrent += current;
    totalOffer += bestOffer;
  });

  const annualSavings = totalCurrent - totalOffer;
  const savingsPercent = totalCurrent > 0 ? (annualSavings / totalCurrent) * 100 : 0;

  const policies: ComparisonPolicyRow[] = rows.map((row: any) => {
    const icon = policyIconForType[row.policyType] ?? <FeatherHome />;
    const current = row.current?.pricing?.annualPremium ?? null;
    const bestOffer = row.offers?.[0]?.pricing?.annualPremium ?? null;
    const savings = current != null && bestOffer != null ? current - bestOffer : null;

    return {
      policyType: row.policyType,
      label: row.policyTypeLabel ?? row.policyType,
      icon,
      currentAnnual: current,
      offerAnnual: bestOffer,
      annualSavings: savings,
      statusLabel: "Se detaljer",
      statusVariant: "success",
    };
  });

  return {
    title: "Dine samlede tilbud",
    subtitle: "Se hvad du kan spare på tværs af alle forsikringer",
    currentCompanyName: "", // could be “Nuværende selskaber”
    offerCompanyName: "",   // could be “Bedste tilbud”
    totalAnnualSavings: annualSavings,
    totalSavingsPercent: savingsPercent,
    totalCurrentAnnual: totalCurrent,
    totalOfferAnnual: totalOffer,
    policies,
    coverageRows: [],
    highlights: [],
    cumulativeSavings: undefined,
    chartData: undefined,
    canOpenMessages: false,
  };
}
```

Then in `client/src/pages/offer-comparison-all.tsx`:

* Turn it into a container that calls `usePolicyComparisons`.
* Transform `data` with `transformPolicyComparisonsToViewModel`.
* Render `<ComparisonPageLayout view={view} />`.

Do **not** keep any hardcoded numbers in this page; all layout should now live in `ComparisonPageLayout`.
