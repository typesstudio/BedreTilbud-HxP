export async function ensureHealthCheckForSnapshot(
  snapshotId: string,
  userId: string
) {
  const { db } = await import("./db");
  const { healthChecks, policySnapshots } = await import("@shared/schema");
  const { eq } = await import("drizzle-orm");

  // 1) Check if one already exists
  const [existing] = await db
    .select()
    .from(healthChecks)
    .where(eq(healthChecks.snapshotId, snapshotId));

  if (existing) return existing;

  // 2) Load the snapshot we’re generating for
  const [snapshot] = await db
    .select()
    .from(policySnapshots)
    .where(eq(policySnapshots.id, snapshotId));

  if (!snapshot) {
    throw new Error(`Snapshot not found for id ${snapshotId}`);
  }

  // 3) Call the existing health-check orchestrator / agent
  //    IMPORTANT: Reuse existing code, do NOT duplicate business logic.
  const { generateHealthCheckForSnapshot } = await import(
    "./HealthCheckOrchestrator" // adjust to actual file
  );

  const healthCheck = await generateHealthCheckForSnapshot({
    snapshot,
    userId,
  });

  // 4) Persist and return
  return healthCheck;
}

Step 4 – Backfill missing health checks for existing snapshots
We need a one-off script to create health checks for snapshots that don’t have one yet (for your current test comparison and others).


Create a script file, e.g. server/scripts/backfill-missing-health-checks.ts.


Logic:


Query all policySnapshots that belong to real users (or simply all for now).


LEFT JOIN to the health-checks table.

For each such snapshot:

Pseudocode:
import "dotenv/config";
import { db } from "../db";
import { policySnapshots, healthChecks } from "@shared/schema";
import { eq, isNull } from "drizzle-orm";
import { ensureHealthCheckForSnapshot } from "../services/healthCheckService";

async function main() {
  const rows = await db
    .select({ id: policySnapshots.id, userId: policySnapshots.userId })
    .from(policySnapshots)
    .leftJoin(
      healthChecks,
      eq(healthChecks.snapshotId, policySnapshots.id)
    )
    .where(isNull(healthChecks.snapshotId));

  console.log(`Found ${rows.length} snapshots without health checks`);

  for (const row of rows) {
    console.log("Generating health check for snapshot", row.id);
    await ensureHealthCheckForSnapshot(row.id, row.userId);
  }

  console.log("Done");
}

main().catch((err) => {
  console.error(err);
  process.exit(1);
});
