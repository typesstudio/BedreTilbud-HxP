### 5. Create hook + transformer for **policy-level health checks**

1. In `client/src/hooks`, create:
   `client/src/hooks/usePolicyHealthCheck.ts`

   ```ts
   import { useQuery } from "@tanstack/react-query";
   import { HealthCheckLayoutProps } from "@/components/health/HealthCheckLayout";

   export function usePolicyHealthCheck(snapshotId: string | undefined) {
     return useQuery({
       queryKey: ["policyHealthCheck", snapshotId],
       enabled: !!snapshotId,
       queryFn: async (): Promise<HealthCheckLayoutProps> => {
         const res = await fetch(`/api/policies/health-check/${snapshotId}`);
         if (!res.ok) throw new Error("Failed to load health check");
         const apiData = await res.json();
         return transformPolicyHealthCheckToView(apiData);
       },
     });
   }
   ```
