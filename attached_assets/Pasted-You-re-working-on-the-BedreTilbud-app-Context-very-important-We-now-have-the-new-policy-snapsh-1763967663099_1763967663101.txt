import { useQuery } from '@tanstack/react-query';

export function usePolicyOverview(snapshotId: string) {
  return useQuery({
    queryKey: ['policy-overview', snapshotId],
    queryFn: async () => {
      const res = await fetch(`/api/policies/${snapshotId}/overview`);
      if (!res.ok) throw new Error('Failed to fetch policy overview');
      return res.json();
    },
  });
}
